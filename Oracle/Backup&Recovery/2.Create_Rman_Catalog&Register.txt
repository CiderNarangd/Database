-- In Catalog Host

// rman catalog tablespace 생성
create tablespace rman_ts datafile '/u02/oradata/ORCL/rman_ts.dbf' size 5000m;

// rman user 생성
create user ruser identified by ruser
default tablespace rman_ts
quota unlimited on rman_ts;

grant connect, resource, recovery_catalog_owner to ruser;

//rman catalog 생성

rman catalog ruser/ruser@ramn_remote
