{
  allocate channel c1 type disk;
  allocate channel c2 type disk;
  allocate auxiliary channel sby type disk;
  allocate auxiliary channel sby2 type disk;
  
  duplicate target database for standby from active database
  DORECOVER
  spfile
    set db_name           = 'master'
    set db_unique_name    = 'slave'
    set audit_file_dest   = '/u01/app/oracle/admin/slave/adump/'
    set diagnostic_dest   = '/u01/app/oracle/'
    set db_create_file_dest = '/u02/oradata/'
    set db_recovery_file_dest
                          = '/u02/oradata/slave/fast_recovery_area'
    set db_recovery_file_dest_size = '15G'
    set log_archive_dest_2 =
      'service=master async
       valid_for=(online_logfile,primary_role)
       db_unique_name=master'
    set log_archive_max_processes = '4'
    set fal_server          = 'master'
    set standby_file_management = 'AUTO'
    set log_archive_config  = 'dg_config=(master,slave)';
 
  sql channel sby
    "alter database recover managed standby database disconnect from session";
}
