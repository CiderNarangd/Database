19C Create Database 

======= Single Instance

//With DBCA

Host - Oracle19c-m
//non-CDB
dbca -silent -createDatabase \
  -templateName General_Purpose.dbc \
  -gdbName ORCL \
  -sid ORCL \
  -responseFile NO_VALUE \
  -characterSet AL32UTF8 \
  -nationalCharacterSet AL16UTF16 \
  -createAsContainerDatabase false \
  -databaseType MULTIPURPOSE \
  -memoryMgmtType AUTO_SGA \
  -totalMemory 3072 \
  -datafileDestination /u02/oradata \
  -recoveryAreaDestination /u02/oradata/ORCL/fast_recovery_area \
  -recoveryAreaSize 10240 \
  -redoLogFileSize 100 \
  -emConfiguration NONE \
  -sysPassword oracle \
  -systemPassword oracle
  
** databaseType // OLTP or MULTIPURPOSE

//Database 생성

vi create_db.sql
=====================================================

CREATE DATABASE ORCL
  USER SYS IDENTIFIED BY oracle
  USER SYSTEM IDENTIFIED BY oracle
  LOGFILE GROUP 1 ('/u02/oradata/ORCL/redo01a.log','/u02/oradata/ORCL/redo01b.log') SIZE 100M,
          GROUP 2 ('/u02/oradata/ORCL/redo02a.log','/u02/oradata/ORCL/redo02b.log') SIZE 100M,
          GROUP 3 ('/u02/oradata/ORCL/redo03a.log','/u02/oradata/ORCL/redo03b.log') SIZE 100M
  MAXLOGFILES 5
  MAXLOGMEMBERS 3
  MAXDATAFILES 100
  CHARACTER SET AL32UTF8
  NATIONAL CHARACTER SET AL16UTF16
  EXTENT MANAGEMENT LOCAL
  DATAFILE '/u02/oradata/ORCL/system01.dbf' SIZE 700M AUTOEXTEND ON NEXT 10M MAXSIZE UNLIMITED
  SYSAUX DATAFILE '/u02/oradata/ORCL/sysaux01.dbf' SIZE 550M AUTOEXTEND ON NEXT 10M MAXSIZE UNLIMITED
  DEFAULT TABLESPACE USERS DATAFILE '/u02/oradata/ORCL/users01.dbf' SIZE 200M AUTOEXTEND ON NEXT 10M MAXSIZE UNLIMITED
  DEFAULT TEMPORARY TABLESPACE TEMP TEMPFILE '/u02/oradata/ORCL/temp01.dbf' SIZE 100M AUTOEXTEND ON NEXT 10M MAXSIZE UNLIMITED
  UNDO TABLESPACE UNDOTBS1 DATAFILE '/u02/oradata/ORCL/undotbs01.dbf' SIZE 200M AUTOEXTEND ON NEXT 10M MAXSIZE UNLIMITED;
