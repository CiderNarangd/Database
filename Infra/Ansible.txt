$ vi Entier_SSH_Register.sh

=========================================

#!/usr/bin/env bash
set -euo pipefail

ORACLE_USER="ansible"
MYSQL_USER="kinam"
PUBKEY=/home/kinam/.sshid_ed25519.pub

oracle_hosts=(
  orcl-19c-si
  orcl-19c-dg-primary
  orcl-19c-dg-standby
  orcl-19c-rac01
  orcl-19c-rac02
  orcl-19c-rac03
  orcl-19c-rman
)

mysql_hosts=(
  mysql-84-replica
  mysql-84-source
  mysql-ibc-router
  mysql-ibc-primary
  mysql-ibc-sec01
  mysql-ibc-sec02
  mysql-84-backup
  Monitoring
)

for host in "${oracle_hosts[@]}"; do
  ssh-copy-id -i "$PUBKEY" "${ORACLE_USER}@${host}"
done

for host in "${mysql_hosts[@]}"; do
  ssh-copy-id -i "$PUBKEY" "${MYSQL_USER}@${host}"
done

// 파일 배포 Playbook 작성

$ vi deploy_file.yaml

=======================================================

- name: Deploy file to mysql servers
  hosts: mysql_replication
  vars:
    src_file: "/home/kinam/ansible_test.txt"
    dest_path: "/storage/downloads/ansible_tests.txt"
    backup_before_copy: false

  tasks:
    - name: 파일 배포
      ansible.builtin.copy:
        src: "{{ src_file }}"
        dest: "{{ dest_path }}"
        owner: kinam
        group: kinam
        mode: '0755'
        backup: yes          
      register: copy_result

	- name: 배포 결과 확인
      ansible.builtin.debug:
        msg: "파일 변경됨: {{ copy_result.changed }}"
	  
========================================================

-- Synatx Check
$ ansible-playbook -i inventory.yaml deploy_file.yaml --syntax-check

-- 실제 적용 전 dry-run
$ ansible-playbook -i inventory.yaml deploy_file.yaml --check --diff

-- 실제 실행
$ ansible-playbook -i inventory.yaml deploy_file.yaml

-- 특정 호스트들만 대상
$ ansible-playbook -i inventory.yaml deploy_file.yaml --limit mysql_replication

-- Ad-hoc Example
$ ansible mysql_replication -i inventory.yaml -m shell -a "df -h /u01"
