$ mv prometheus-3.13.1.linux-amd64 prometheus
$ mv prometheus /usr/local

//Prometheus 데이터 디렉토리 생성

$ mkdir -p /u01/prometheus/data


//환경변수 추가
-- bash profile 추가
$ vi .bash_profile
==================================================

export PROMETHEUS_HOME=/usr/local/prometheus
export PATH=$PROMETHEUS_HOME

==================================================

-- bashrc alias 추가
$ vi .bashrc
=============================================

alias prom='cd ${PROMETHEUS_HOME}'
alias promdata='cd /u01/prometheus/data/'

=============================================


// Systemctl 등록

vi /etc/systemd/system/prometheus.service

===============================================

[Unit]
Description=Prometheus Server
After=network-online.target

[Service]
User=kinam
Restart=on-failure
ExecStart=/usr/local/prometheus/prometheus \
  --config.file=/usr/local/prometheus/prometheus.yml \
  --storage.tsdb.path=/u01/prometheus/data/

[Install]
WantedBy=multi-user.target

==================================================================

$ systemctl daemon-reload
$ systemctl start prometheus
$ systemctl status prometheus
$ systemctl stop prometheus
$ systemctl enable prometheus
