Vagrantfile 생성

========================================

Vagrant.configure("2") do |config|

  config.vm.box = "rocky9-mysql-base"
  config.ssh.username = "kinam"
  config.ssh.password = "oracle"
  
  config.ssh.host = "192.168.56.205"
  config.ssh.port = 22  
  
  config.vm.provider "virtualbox" do |vb|
    
    vb.customize ["modifyvm", :id, "--nic1", "hostonly"]
    vb.customize ["modifyvm", :id, "--hostonlyadapter1", "VirtualBox Host-Only Ethernet Adapter"]
    
    vb.customize ["modifyvm", :id, "--nic2", "nat"]

  end
	
  # host-only 고정할 ip 설정	
  config.vm.network "private_network", ip: "192.168.56.205", adapter: 1
  
end

Vagrant Multi Deploy

cd E:\Git2\Database\Infra\my_vagrant\mysql-replication


Vagrantfile 생성
========================================================

Vagrant.configure("2") do |config|

  # 공통 설정
  boxes = [
    { name: "mysql-master", ip: "192.168.56.210" },
    { name: "mysql-slave", ip: "192.168.56.211" },
  ]

  boxes.each do |box|
    config.vm.define box[:name] do |node|
      node.vm.box      = "rocky9-mysql-base"
      node.vm.hostname = box[:name]

      node.ssh.username = "kinam"
      node.ssh.password = "oracle"

      node.vm.provider "virtualbox" do |vb|
        vb.name   = box[:name]
        vb.memory = 2048
        vb.cpus   = 2
        vb.customize ["modifyvm", :id, "--nic1", "nat"]
        vb.customize ["modifyvm", :id, "--nic2", "hostonly"]
        vb.customize ["modifyvm", :id, "--hostonlyadapter2", "VirtualBox Host-Only Ethernet Adapter"]
      end

      node.vm.network "private_network", ip: box[:ip], adapter: 2

    end
  end

end
