// === Loki write endpoint ===
loki.write "default" {
  endpoint {
    url = "http://192.168.56.52:3100/loki/api/v1/push"
  }
}

// === MySQL slow query log ===
local.file_match "mysql_slowlog" {
  path_targets = [{
    __path__ = "/u01/mysql/mysqltemplate/DATA/mysql-84-source-slow.log",
    job      = "mysql_slowlog",
    host     = constants.hostname,
  }]
}

loki.source.file "mysql_slowlog" {
  targets    = local.file_match.mysql_slowlog.targets
  forward_to = [loki.process.mysql_slowlog.receiver]
}

loki.process "mysql_slowlog" {
  stage.multiline {
    firstline     = "^# Time: "
    max_wait_time = "3s"
  }

  stage.regex {
    expression = "(?s)# Time: (?P<time>\\S+)\\n# User@Host: (?P<user_host>[^\\n]*)\\n# Query_time: (?P<query_time>[0-9.]+)\\s+Lock_time: (?P<lock_time>[0-9.]+)\\s+Rows_sent: (?P<rows_sent>[0-9]+)\\s+Rows_examined: (?P<rows_examined>[0-9]+)"
  }
  
  stage.timestamp {
    source = "time"
    format = "2006-01-02T15:04:05.999999999Z"
  }

  stage.labels {
    values = {
      job  = "",
      host = "",
    }
  }

  stage.structured_metadata {
    values = {
      query_time    = "",
      lock_time     = "",
      rows_sent     = "",
      rows_examined = "",
      user_host     = "",
    }
  }

  forward_to = [loki.write.default.receiver]
}
