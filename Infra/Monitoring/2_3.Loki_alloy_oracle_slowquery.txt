#!/bin/bash
export ORACLE_SID="${1:?Usage: $0 <ORACLE_SID>}"
export ORACLE_HOME=/u01/app/oracle/product/19.3.0/dbhome_1 
export PATH=$ORACLE_HOME/bin:$PATH
 
OUT_DIR=/storage/downloads
OUT_FILE="${OUT_DIR}/${ORACLE_SID}-SlowQuery.log"
 
mkdir -p "$OUT_DIR"
 
sqlplus -s prometheus/oracle <<SQL >> "$OUT_FILE"
SET PAGESIZE 0 LINESIZE 2000 FEEDBACK OFF HEADING OFF TRIMSPOOL ON TERMOUT OFF ECHO OFF
SET TIMING OFF TIME OFF VERIFY OFF AUTOTRACE OFF SQLPROMPT ''
SET NEWPAGE NONE

SELECT 'time="'   || TO_CHAR(SYSTIMESTAMP,'YYYY-MM-DD"T"HH24:MI:SSTZH:TZM') || '" ' ||
       'oracle_sid="' || SYS_CONTEXT('USERENV','INSTANCE_NAME') || '" ' ||
       'sql_id="' || sql_id || '" ' ||
       'schema="' || NVL(parsing_schema_name,'-') || '" ' ||
       'avg_time_sec="' || ROUND(elapsed_time / executions / 1000000, 3) || '" ' ||
       'executions="' || executions || '" ' ||
       'last_active="' || TO_CHAR(last_active_time,'YYYY-MM-DD"T"HH24:MI:SS') || '" ' ||
       'sql_text="' || REPLACE(REPLACE(SUBSTR(sql_text,1,300),CHR(10),' '),'"','''') || '"'
FROM v\$sql
WHERE executions > 0
  AND last_active_time >= SYSDATE - (5/1440)		# 스크립트 기준 수집할 시간값
  AND parsing_schema_name NOT IN
      ('SYS','SYSTEM','DBSNMP','OUTLN','APPQOSSYS','DBSFWUSER','GGSYS',
       'XS\$NULL','WMSYS','ORDSYS','ORDDATA','CTXSYS','MDSYS','OLAPSYS',
       'GSMADMIN_INTERNAL','MONUSER')
  AND elapsed_time / executions / 1000000 >= 1;		# slow query 임계치값
 
EXIT;
SQL

// === Loki write endpoint ===
loki.write "default" {
  endpoint {
    url = "http://192.168.56.52:3100/loki/api/v1/push"
  }
}

// === Oracle slow query log ===
local.file_match "oracle_slowlog" {
  path_targets = [{
    __path__ 	= "/storage/downloads/orcl-SlowQuery.log",
    job      	= "oracle_slowlog",
    host     	= constants.hostname,
	oracle_sid  = "orcl",
  }]
}

loki.source.file "oracle_slowlog" {
  targets    = local.file_match.oracle_slowlog.targets
  forward_to = [loki.process.oracle_slowlog.receiver]
}

loki.process "oracle_slowlog" {
  stage.logfmt {
    mapping = {
      "time"         = "",
      "oracle_sid"   = "",
      "sql_id"       = "",
      "schema"       = "",
      "avg_time_sec" = "",
      "executions"   = "",
      "last_active"  = "",
      "sql_text"     = "",
    }
  }

  stage.timestamp {
    source = "time"
    format = "2006-01-02T15:04:05Z07:00"
  }

  stage.labels {
    values = {
      job        = "",
      host       = "",
      oracle_sid = "",
      schema     = "",
    }
  }

  stage.structured_metadata {
    values = {
      sql_id       = "",
      avg_time_sec = "",
      executions   = "",
      last_active  = "",
      sql_text     = "",
    }
  }

  forward_to = [loki.write.default.receiver]
}
