 -- 아래 rule 설정 수정
===================================================/

alerting:
  alertmanagers:
    - static_configs:
        - targets:
            - "192.168.56.52:9093"

rule_files:
  - "/usr/local/prometheus/rule_node.yml"
  - "/usr/local/prometheus/rule_mysqld.yml"
  - "/usr/local/prometheus/rule_oracle.yml"
  
===================================================/
