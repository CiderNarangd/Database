  database {
	name              = "orcl-19c-si"
	connection_string = "oracle://prometheus:oracle@orcl_19c_si"
  }
  #custom_metrics = ["/etc/alloy/oracle_dataguard_metrics.toml"]
  
  or 
  
  database {
	name              = "orcl-19c-si"	# 프로메테우스에서 식별할 이름
