#!/usr/bin/env bash

BUCKET_NAME=s3://kinam-s3-mysqlbackup/mysql-backup
HOST="$(hostname -s)"
DATE=$(date +%Y%m%d)
BACKUP_DIR=/u01/mysql/${HOST}/BACKUP/${DATE}_full
BUCKET_DIR=${BUCKET_NAME}/${DATE}

if [ ! -d ${BACKUP_DIR} ] ; then
	echo "No Backup Exists : ${BACKUP_DIR}"
	exit 0
fi

echo "${DATE} backup file upload start to s3"
if aws s3 sync "${BACKUP_DIR}" "${BUCKET_DIR}"; then
    echo "${DATE} backup upload success"
else
    echo "${DATE} backup upload FAILED" >&2
    exit 1
fi
