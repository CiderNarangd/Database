Loki Install

OS 				- Rocky 908
Loki Version 	- 3.7.2
Port 			- 3100

$ unzip loki-linux-amd64.zip 
$ ./loki-linux-amd64 --version
loki, version 3.7.2 (branch: release-3.7.x, revision: 7486c4a7)
  build user:       root@623d39919632
  build date:       2026-05-13T08:54:45Z
  go version:       go1.26.2
  platform:         linux/amd64
  tags:             netgo
  
$ mv loki-linux-amd64 loki
$ mv loki /usr/local/bin

// 로키 데이터 디렉토리 생성
$ mkdir -p /u01/loki/data
$ mkdir -p /u01/loki/data/chunks
$ mkdir -p /u01/loki/data/rules
$ mkdir -p /u01/loki/data/compactor

// Loki Config Downlaod
$ wget https://raw.githubusercontent.com/grafana/loki/main/cmd/loki/loki-local-config.yaml

$ vi loki-local-config.yaml
$ mv loki-local-config.yaml /etc/loki_config.yaml

// systemctl 등록
$ vi /etc/systemd/system/loki.service
===========================================================================

[Unit]
Description=Loki
After=network-online.target

[Service]
User=root
Restart=on-failure
ExecStart=/usr/local/bin/loki -config.file /etc/loki_config.yaml

[Install]
WantedBy=multi-user.target

===========================================================================

$ systemctl daemon-reload
$ systemctl start loki
$ systemctl status loki
$ systemctl stop loki
$ systemctl enable loki
