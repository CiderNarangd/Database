Postregsql Archive Mode



$ su postgres

-- db 접속 후 configle 위치 조회
postgres=# show config_file;
                 config_file                  
----------------------------------------------
 /u01/pgsql/pgsql-master/data/postgresql.conf
(1 row)


// archive 디렉토리 생성

$ pwd
/u01/pgsql/pgsql-master
$ ll
total 4
drwx------ 19 postgres postgres 4096 Aug 11 23:00 data
drwxrwxr-x  2 postgres postgres   28 Aug  4 16:46 log

$ mkdir -p archive


$ vi /u01/pgsql/pgsql-master/data/postgresql.conf

========================================================

wal_level 			= replica  	## 최소 replica 이상 필요
archive_mode 		= on		## 재시작 필요함
archive_timeout 	= 300		## 5분으로 설정
archive_command 	= 'test ! -f /u01/pgsql/pgsql-master/archive/arch%f && cp %p /u01/pgsql/pgsql-master/archive/arch%f'

=====================================================


// 재시작

$ systemctl restart pgsql
or 
-- wal_mode 변경안했을시는 config reload만
postgres=# SELECT pg_reload_conf();

postgres=# SHOW archive_command;
