///// PostgreSQL 17 Install ////

OS - Oralce Linux 9.4
2core 4gb ram
DB v - Postgre 17.7 


//시스템 업데이트
dnf update -y

dnf groupinstall -y "Development Tools"
dnf install -y \
    readline-devel \
    zlib-devel \
    libxml2-devel \
    libxslt-devel \
    openssl-devel \
    systemd-devel \
    perl-ExtUtils-Embed \
    python3-devel \
    tcl-devel \
    openldap-devel \
    pam-devel \
    perl-FindBin \	
    libicu-devel \
    llvm-devel \
    clang

dnf install -y perl-FindBin
dnf install -y bison
dnf install -y flex

//소스 다운로드
cd /storage/download
wget https://ftp.postgresql.org/pub/source/v17.7/postgresql-17.7.tar.gz
tar -xvf  postgresql-17.7.tar.gz


//pgsql 그룹,계정 생성
useradd -m -d /home/postgres -s /bin/bash postgres


//pgsql 데이터, 로그 디렉토리 생성
mkdir -p /usr/local/pgsql17
mkdir /pgdata
mkdir /pglog

// 디렉토리 권한 변경
chown -R postgres:postgres /usr/local/pgsql17
chown -R postgres:postgres /pgdata
chown -R postgres:postgres /pglog

//컴파일 및 설치

cd /storage/download/postgresql-17.7
./configure --prefix=/usr/local/pgsql17
make -j$(nproc)
make install


//사용자 전환후 db초기화
su - postgres
cd /usr/local/pgsql17/bin/
./initdb -D /pgdata

//db서버시작
su postgres
/usr/local/pgsql17/bin/pg_ctl -D /pgdata -l /pglog/logfile start

//Path등록
vi ~/.bashrc

export PGHOME=/usr/local/pgsql17
export PATH=$PGHOME/bin:$PATH

source ~/.bashrc


//systemd 등록
su root
vi /etc/systemd/system/postgresql.service

========================================
[Unit]
Description=PostgreSQL 17 database server
After=network.target

[Service]
Type=forking
User=postgres
Group=postgres
ExecStart=/usr/local/pgsql17/bin/pg_ctl start -D /pgdata -l /pgdata/logfile
ExecStop=/usr/local/pgsql17/bin/pg_ctl stop -D /pgdata
ExecReload=/usr/local/pgsql17/bin/pg_ctl reload -D /pgdata

[Install]
WantedBy=multi-user.target
========================================================

systemctl daemon-reload
systemctl enable postgresql
systemctl start postgresql

//포트확인
ss -lntp | grep 5432

//외부 접속을 위한 방화벽 설정
firewall-cmd --zone=public --add-port=5432/tcp --permanent
firewall-cmd --reload
firewall-cmd --list-ports
