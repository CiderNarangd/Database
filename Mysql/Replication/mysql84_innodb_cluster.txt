###### Router Install ######

-- In mysql-84-router Hosts

$ tar -xvf mysql-router-8.4.8-linux-glibc2.28-x86_64.tar.xz
$ mv mysql-router-8.4.8-linux-glibc2.28-x86_64 /usr/local
$ cd /usr/local
$ chown -R mysql:mysql mysql-router-8.4.8-linux-glibc2.28-x86_64
$ ln -s mysql-router-8.4.8-linux-glibc2.28-x86_64 mysql-router

$ vi .bash_profile

##### Cluster 구성 #####

// 계정 생성

-- On Primary,Secondary01,02

* 편의상 전체 권한 & 대역대 설정
* 문서 하단 실무용 권한 설정 확인

mysql> CREATE USER 'cluster_admin'@'%' IDENTIFIED BY 'oracle';
mysql> GRANT ALL PRIVILEGES ON *.* TO 'cluster_admin'@'%' WITH GRANT OPTION;
mysql> flush privileges;


// 접속 확인

$ mysqlsh cluster_admin@mysql-ibc-primary:3306 --js
$ mysqlsh cluster_admin@mysql-ibc-sec01:3306 --js
$ mysqlsh cluster_admin@mysql-ibc-sec02:3306 --js

$ mysqlsh cluster_admin@localhost:3306 --js

* 패스워드 저장 y

-- Pre-Checking Instance 
-- 각 호스트에서 진행

js> dba.checkInstanceConfiguration('cluster_admin@mysql-ibc-primary:3306');
js> dba.checkInstanceConfiguration('cluster_admin@mysql-ibc-sec01:3306');
js> dba.checkInstanceConfiguration('cluster_admin@mysql-ibc-sec02:3306');

* 아래 configureInstance() 함수 사용하면서 해결됨.

-- 각 노드 호스트에서
$ mysqlsh cluster_admin@127.0.0.1:3306 --js
js > dba.configureInstance();


-- 모든 노드의 uuid값 확인
sql> show global variables like '%server_uuid%';
+---------+---------------+-----------------------------------------+
|primary - | server_uuid   | d472d103-79cf-11f1-a1ce-080027dbb89f   |
|sec01   - | server_uuid   | dd3a3c02-79cf-11f1-96e4-080027dbb89f   |

** innodb cluster에선 uuid 기반으로 작동한다. 하지만 binlog 기반 작업 및 다른 리플리케이션 연결에 대비하여
   server-id값도 고유하게 변경해주도록 한다
   
   primary 	- 1
   sec01	- 2
   sec02	- 3
