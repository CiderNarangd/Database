// Router Port Description (Default Set)
- Innodb에선 라우터를 사용하여 로드밸런싱 사용가능
- 6446 - Primary / Read&Write		// 일반적으로 이곳으로 커넥션 붙어서 작업
- 6447 - Secondarys / Read only		// 읽기전용으로 read 전용 커넥션 부하 분산 가능 ( 배치,통계 처리등 )
- 6448 - Mysql Shell Document Store (X Protocol Read&Write)
- 6449 - Document store read (X Protocol Read Only) 

- 6446,6447 포트만 알면 크게 문제 없을듯 하다.

// 실무용 클러스터 계정 권한 설정
GRANT 
  CONNECTION_ADMIN,
  GROUP_REPLICATION_ADMIN,
  PERSIST_RO_VARIABLES_ADMIN,
  SYSTEM_VARIABLES_ADMIN,
  BACKUP_ADMIN,
  CLONE_ADMIN,
  PROCESS, 
  RELOAD, 
  REPLICATION SLAVE, 
  REPLICATION_APPLIER,
  REPLICATION CLIENT
ON *.* TO 'cluster_admin'@'%';

GRANT SELECT ON mysql.* TO 'cluster_admin'@'%';
GRANT SELECT ON performance_schema.* TO 'cluster_admin'@'%';
GRANT SELECT ON sys.* TO 'cluster_admin'@'%';

FLUSH PRIVILEGES;

// 함수들..
var cluster = dba.createCluster("kinamCluster");
var cluster = dba.getCluster();
cluster.status();
cluster.status({extended: 1});
cluster.describe();
cluster.addInstance("cluster_admin@mysql-ibc-sec01:3306");
cluster.removeInstance("mysql-ibc-sec01:3306",{force:true});
cluster.rejoinInstance("mysql-ibc-sec01:3306");
cluster.rescan();
cluster.setPrimaryInstance("mysql-ibc-sec01:3306");
cluster.options();
cluster.setOption("exitStateAction","READ_ONLY");
cluster.disconnect();
