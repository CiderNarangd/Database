Mysql Replication 

DB Version - 8.4.8
OS Version - Rocky 9.8

Source  host - mysql-84-source  // 192.168.56.201
Replica host - mysql-84-replica // 192.168.56.202


// Modify mysql server-id

$ vi /etc/my.cnf
-- In Source
 server-id=1

-- In Replica
 server-id=2
 read-only = 1 -- 필요시 설정

// Create Replication User in Source
create user 'replica'@'%' identified by 'oracle';
grant replication slave on *.* to 'replica'@'%'; 	
flush privileges;


// Perform a full backup on the Source host

-- Mkdir Backup dir in source,replica
$ mkdir -p /home/kinam/backup
$ chmod -R 775 /home/kinam/backup

-- Progress backup
$ xtrabackup \
	--defaults-file=/etc/my.cnf \
	--backup \
    --host=localhost \
    --user=xtrabackup \
    --password='oracle' \
	--no-lock \
	--parallel=4 \
	--no-version-check \
    --target-dir=/home/kinam/backup
	
$ xtrabackup --prepare --parallel=4 --target-dir=/home/kinam/backup

-- Transfer backup to replica
$ scp -r /home/kinam/backup kinam@mysql-84-replica:/home/kinam/backup

-- In replica change owner
$ chown -R mysql:mysql /home/kinam/backup


// Restore in Replica
-- Directory 삭제
$ rm -rf /u01/mysql/${HOST_NAME}/DATA/*
$ rm -rf /u01/mysql/${HOST_NAME}/IBLOG/*
$ rm -rf /u01/mysql/${HOST_NAME}/BINLOG/*

-- Copy-Back 
$ xtrabackup \
	--defaults-file=/etc/my.cnf \
	--copy-back \
	--parallel=4 \
	--target-dir=/home/kinam/backup
	
$ chown -R mysql:mysql /u01/mysql/${HOST_NAME}/DATA
$ chown -R mysql:mysql /u01/mysql/${HOST_NAME}/IBLOG
$ chown -R mysql:mysql /u01/mysql/${HOST_NAME}/BINLOG

$ systemctl start mysqld
