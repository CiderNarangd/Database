-- Command & Trouble Shooting

--> gtid 기반에서는 skil_counter =1; 이 작동하지 않아서 아래 방법 사용.

1.
-- 1. 슬레이브 복제 중지
STOP SLAVE;

-- 2. 문제가 된 GTID 번호를 수동으로 실행한 것처럼 세션에 설정
SET GTID_NEXT = '1a2b3c4d-5e6f-7g8h-9i0j-1234567890ab:100';

-- 3. 내용이 없는 빈 트랜잭션 시작 및 커밋
BEGIN; COMMIT;

-- 4. GTID 설정을 다시 자동(Automatic)으로 원복 (필수!)
SET GTID_NEXT = 'AUTOMATIC';

-- 5. 슬레이브 복제 다시 시작
START SLAVE;
