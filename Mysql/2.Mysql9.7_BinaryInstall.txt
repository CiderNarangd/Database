ln -s /usr/local/mysql-9.7.0-linux-glibc2.28-x86_64 mysql

//mysql Log & Data dir생성

Log DIr	 - /u01/mysql/{Host_name}/ADMIN
Bin Log  - /u01/mysql/{Host_name}/BINLOG
IBLOG	 - /u01/mysql/{Host_name}/IBLOG

mkdir -p  /u01/mysql/mysql97-m/DATA

//My.cnf 작성 
vi /etc/my.cnf

===============================

[client]
port            = 3306
socket          = /u01/mysql/mysql97-m/DATA/mysql.sock
#password       = your_password

[mysqld]
#data dir
basedir         = /usr/local/mysql
datadir         = /u01/mysql/mysql97-m/DATA
socket          = /u01/mysql/mysql97-m/DATA/mysql.sock

log-error       = /u01/mysql/mysql97-m/ADMIN/error_log.log

innodb_log_group_home_dir = /u01/mysql/mysql97-m/IBLOG
innodb_data_home_dir      = /u01/mysql/mysql97-m/DATA

===========================

chown -R mysql:mysql /etc/my.cnf
chmod 775 -R /etc/my.cnf


//initializing
/usr/local/mysql/bin/mysqld --defaults-file=/etc/my.cnf --initialize --user=mysql

// systemd 등록
1.
cp /usr/local/mysql/support-files/mysql.server /etc/init.d/mysqld
chmod +x /etc/init.d/mysqld
mysql.server 스크립트 내부의 값 입력 필요
=====================
basedir=/usr/local/mysql
datadir=/u01/mysql/mysql97-m/DATA
=======================

2.
sudo vi /etc/systemd/system/mysqld.service
==============================
[Unit]
Description=MySQL Server
After=network.target

[Service]
User=mysql
Group=mysql
Type=notify
ExecStart=/usr/local/mysql/bin/mysqld --defaults-file=/etc/my.cnf
LimitNOFILE=65535

[Install]
WantedBy=multi-user.target
====================================

sudo systemctl daemon-reload
sudo systemctl enable mysqld
sudo systemctl start mysqld
sudo systemctl stop mysqld
sudo systemctl status mysqld
